    private void ShowGuidePanel()
    {
        if (guideOverlay == null || guidePanel == null || guideSlides.Count == 0)
        {
            return;
        }

        guideIndex = 0;
        ApplyGuideSlide(guideIndex);

        guideOverlay.style.display = DisplayStyle.Flex;
        guideOverlay.BringToFront();
        guidePanel.RemoveFromClassList("show-guide-panel");
        guidePanel.schedule.Execute(() =>
        {
            guidePanel.AddToClassList("show-guide-panel");
        }).StartingIn(GuidePanelTransitionDelayMs);
    }

    private void OnGuideNextClicked(ClickEvent evt)
    {
        if (guideSlides.Count == 0)
        {
            return;
        }

        if (guideIndex >= guideSlides.Count - 1)
        {
            HideGuidePanel();
            if (guideOverlay != null)
            {
                guideOverlay.schedule.Execute(() =>
                {
                    ShowInitialBahanChoice();
                }).StartingIn(GuidePanelHideDelayMs);
            }
            return;
        }

        guideIndex++;
        ApplyGuideSlide(guideIndex);
    }

    private void ApplyGuideSlide(int slideIndex)
    {
        if (slideIndex < 0 || slideIndex >= guideSlides.Count)
        {
            return;
        }

        GuideSlideData slide = guideSlides[slideIndex];

        if (guideCounter != null)
        {
            guideCounter.text = "" +(slideIndex + 1) + " / " + GuideSlideCount;
        }

        if (guideTitle != null)
        {
            guideTitle.text = slide.Title;
        }

        if (guideImageLabel != null)
        {
            guideImageLabel.text = slide.ImageLabel;
        }

        if (guideBodyText != null)
        {
            guideBodyText.text = slide.Body;
        }

        if (guideImageFrame != null)
        {
            guideImageFrame.RemoveFromClassList("guide-image-initial-bahan");
            guideImageFrame.RemoveFromClassList("guide-image-initial-target");
            if (!string.IsNullOrWhiteSpace(slide.ImageClass))
            {
                guideImageFrame.AddToClassList(slide.ImageClass);
            }
        }

        if (guideNextButton != null)
        {
            guideNextButton.text = slideIndex == guideSlides.Count - 1 ? "Start" : "Next";
        }
    }

    private void HideGuidePanel()
    {
        if (guideOverlay == null || guidePanel == null)
        {
            return;
        }

        guidePanel.RemoveFromClassList("show-guide-panel");
        guideOverlay.schedule.Execute(() =>
        {
            guideOverlay.style.display = DisplayStyle.None;
        }).StartingIn(GuidePanelHideDelayMs);
    }
