Table users {
  id uuid [primary key]
  username varchar(50) [not null, unique]
  password_hash varchar(255) [not null, note: 'Store hashed password only, never plain password']
  created_at timestamp
  updated_at timestamp
}

Table players {
  id uuid [primary key]
  user_id uuid [not null]
  name varchar(8) [not null]
}

Table game_sessions {
  id uuid [primary key]
  user_id uuid [not null]
  player_count integer [not null, note: 'Allowed value: 3 or 4']
  day integer [not null, default: 1]
  turn integer [not null, default: 1]
  moves_left integer [not null, default: 2]
  finish_day integer [not null, default: 13]
  is_game_over boolean [not null, default: false]
  created_at timestamp
  updated_at timestamp
}

Table session_players {
  id uuid [primary key]
  session_id uuid [not null]
  player_id uuid [not null]
  player_index integer [not null, note: 'Player order inside session, 1 to 4']
  name varchar(8) [not null, note: 'Snapshot of player name when session starts']
  coins integer [not null, default: 20]
  happiness integer [not null, default: 0]
  saving integer [not null, default: 0]

  indexes {
    (session_id, player_index) [unique]
  }
}

Table bahan_makanan {
  id varchar(80) [primary key]
  nama varchar(80) [not null]
  harga_beli integer [not null]
}

Table resep {
  id varchar(80) [primary key]
  nama varchar(80) [not null]
  harga_jual integer [not null]
  poin_kebahagiaan integer [not null]
}

Table resep_bahan {
  resep_id varchar(80) [not null]
  bahan_id varchar(80) [not null]
  jumlah integer [not null, default: 1]

  indexes {
    (resep_id, bahan_id) [pk]
  }
}

Table kebutuhan {
  id varchar(80) [primary key]
  nama varchar(80) [not null]
  tipe varchar(30) [not null, note: 'primer, sekunder, or tersier']
}

Table tujuan_finansial {
  id varchar(80) [primary key]
  nama varchar(80) [not null]
  harga_beli integer [not null]
  poin_kebahagiaan integer [not null]
}

Table target_kebutuhan {
  id varchar(120) [primary key]
  nama varchar(120) [not null]
  poin_kebahagiaan_berhasil integer [not null]
  poin_kebahagiaan_gagal integer [not null, note: 'Stored as negative value']
}

Table target_kebutuhan_items {
  target_kebutuhan_id varchar(120) [not null]
  urutan integer [not null, note: '1 to 3']
  jenis_target varchar(20) [not null, note: 'tipe or nama']
  value varchar(80) [not null]

  indexes {
    (target_kebutuhan_id, urutan) [pk]
  }
}

Table narasi {
  id varchar(120) [primary key]
  narasi_1 text [not null]
  narasi_2 text
  narasi_3 text
}

Table narasi_prerequisite_aksi {
  id uuid [primary key]
  narasi_id varchar(120) [not null]
  aksi varchar(80) [not null]
  value integer [not null, note: '0 means default narrative for that action']

  indexes {
    (aksi, value)
  }
}

Table quest {
  id varchar(120) [primary key]
  title varchar(120) [not null]
  description text
  aksi varchar(80) [not null]
  target integer [not null]
}

Table player_bahan {
  session_player_id uuid [not null]
  bahan_id varchar(80) [not null]
  jumlah integer [not null, default: 0]

  indexes {
    (session_player_id, bahan_id) [pk]
  }
}

Table player_kebutuhan {
  session_player_id uuid [not null]
  kebutuhan_id varchar(80) [not null]
  tipe varchar(30) [not null]
}

Table player_target_kebutuhan {
  session_player_id uuid [not null]
  target_kebutuhan_id varchar(120) [not null]
  is_completed boolean [not null, default: false]
  is_failed boolean [not null, default: false]
  reward_applied boolean [not null, default: false]

  indexes {
    (session_player_id, target_kebutuhan_id) [pk]
  }
}

Table player_tujuan_finansial {
  session_player_id uuid [not null]
  tujuan_finansial_id varchar(80) [not null]
  purchased_at_day integer [not null]

  indexes {
    (session_player_id, tujuan_finansial_id) [pk]
  }
}

Table player_quest_progress {
  session_player_id uuid [not null]
  quest_id varchar(120) [not null]
  progress integer [not null, default: 0]
  target integer [not null]
  is_completed boolean [not null, default: false]
  is_reward_claimed boolean [not null, default: false]

  indexes {
    (session_player_id, quest_id) [pk]
  }
}

Table player_action_counters {
  session_player_id uuid [not null]
  aksi varchar(80) [not null]
  count integer [not null, default: 0]

  indexes {
    (session_player_id, aksi) [pk]
  }
}

Table player_peduli_donasi {
  session_player_id uuid [primary key]
  total_donasi integer [not null, default: 0]
}

Table peduli_donasi_events {
  id uuid [primary key]
  session_id uuid [not null]
  event_ke integer [not null]
  day integer [not null]

  indexes {
    (session_id, event_ke) [unique]
  }
}

Table peduli_donasi_rankings {
  event_id uuid [not null]
  rank integer [not null]
  session_player_id uuid [not null]
  total_donasi integer [not null]

  indexes {
    (event_id, rank) [pk]
  }
}
